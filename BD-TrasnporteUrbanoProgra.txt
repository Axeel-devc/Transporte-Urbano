/* ============================================================
   TPI - BASE DE DATOS TRANSPORTE_URBANO (SQL Server)
   Parte 1: creacion de la base y las tablas
   Parte 2: carga de datos (incluye las tablas nuevas)
   Parte 3: consultas
   ============================================================ */

CREATE DATABASE Transporte_Urbano;
GO

USE Transporte_Urbano;
GO

/* ============================================================
   PARTE 1 - CREACION DE TABLAS
   ============================================================ */

/* 1. CATALOGOS BASE */

CREATE TABLE Choferes (
    Codigo_Chofer INT IDENTITY(1,1) NOT NULL,
    Nombre VARCHAR(60) NOT NULL,
    Apellido VARCHAR(60) NOT NULL,
    DNI VARCHAR(15) NOT NULL,
    CONSTRAINT PK_Choferes PRIMARY KEY (Codigo_Chofer),
    CONSTRAINT UQ_Choferes_DNI UNIQUE (DNI)
);
GO

CREATE TABLE Calles (
    Codigo_Calle INT IDENTITY(1,1) NOT NULL,
    Nombre_Calle VARCHAR(100) NOT NULL,
    CONSTRAINT PK_Calles PRIMARY KEY (Codigo_Calle)
);
GO

CREATE TABLE Sentidos_Viaje (
    Codigo_Sentido INT IDENTITY(1,1) NOT NULL,
    Sentido VARCHAR(50) NOT NULL,
    CONSTRAINT PK_Sentidos_Viaje PRIMARY KEY (Codigo_Sentido)
);
GO

CREATE TABLE Tipos_Boleto (
    Codigo_Tipo_Boleto INT IDENTITY(1,1) NOT NULL,
    Descripcion VARCHAR(100) NOT NULL,
    Precio DECIMAL(10,2) NOT NULL,
    CONSTRAINT PK_Tipos_Boleto PRIMARY KEY (Codigo_Tipo_Boleto)
);
GO

CREATE TABLE Tipos_Mantenimiento (
    Codigo_Tipo_Mantenimiento INT IDENTITY(1,1) NOT NULL,
    Descripcion VARCHAR(100) NOT NULL,
    CONSTRAINT PK_Tipos_Mantenimiento PRIMARY KEY (Codigo_Tipo_Mantenimiento)
);
GO

/* 2. NUEVAS TABLAS DE CATALOGOS */

CREATE TABLE Estados_Colectivo (
    Codigo_Estado INT IDENTITY(1,1) NOT NULL,
    Descripcion VARCHAR(50) NOT NULL,
    CONSTRAINT PK_Estados_Colectivo PRIMARY KEY (Codigo_Estado),
    CONSTRAINT UQ_Estados_Colectivo_Descripcion UNIQUE (Descripcion)
);
GO

CREATE TABLE Formas_Pago (
    Codigo_Forma_Pago INT IDENTITY(1,1) NOT NULL,
    Descripcion VARCHAR(50) NOT NULL,
    CONSTRAINT PK_Formas_Pago PRIMARY KEY (Codigo_Forma_Pago),
    CONSTRAINT UQ_Formas_Pago_Descripcion UNIQUE (Descripcion)
);
GO

CREATE TABLE Tipos_Repuesto (
    Codigo_Tipo_Repuesto INT IDENTITY(1,1) NOT NULL,
    Descripcion VARCHAR(100) NOT NULL,
    CONSTRAINT PK_Tipos_Repuesto PRIMARY KEY (Codigo_Tipo_Repuesto)
);
GO

CREATE TABLE Proveedores (
    Codigo_Proveedor INT IDENTITY(1,1) NOT NULL,
    Nombre VARCHAR(100) NOT NULL,
    Telefono VARCHAR(25) NULL,
    CONSTRAINT PK_Proveedores PRIMARY KEY (Codigo_Proveedor)
);
GO

/* 3. LINEAS Y COLECTIVOS */

CREATE TABLE Lineas (
    Codigo_Linea INT IDENTITY(1,1) NOT NULL,
    Numero_Linea VARCHAR(15) NOT NULL,
    CONSTRAINT PK_Lineas PRIMARY KEY (Codigo_Linea)
);
GO

CREATE TABLE Colectivos (
    Nro_Coche INT IDENTITY(1,1) NOT NULL,
    Patente VARCHAR(10) NOT NULL,
    Marca VARCHAR(50) NULL,
    Modelo VARCHAR(50) NULL,
    Codigo_Linea INT NOT NULL,
    Codigo_Estado INT NOT NULL,
    CONSTRAINT PK_Colectivos PRIMARY KEY (Nro_Coche),
    CONSTRAINT UQ_Colectivos_Patente UNIQUE (Patente),
    CONSTRAINT FK_Colectivos_Lineas FOREIGN KEY (Codigo_Linea)
        REFERENCES Lineas (Codigo_Linea),
    CONSTRAINT FK_Colectivos_Estados FOREIGN KEY (Codigo_Estado)
        REFERENCES Estados_Colectivo (Codigo_Estado)
);
GO

/* 4. CHOFERES Y ASIGNACIONES */

CREATE TABLE Asignaciones (
    Codigo_Asignacion INT IDENTITY(1,1) NOT NULL,
    Codigo_Chofer INT NOT NULL,
    Nro_Coche INT NOT NULL,
    Fecha DATE NOT NULL,
    Hora_inicio TIME(0) NOT NULL,
    Hora_Fin TIME(0) NOT NULL,
    CONSTRAINT PK_Asignaciones PRIMARY KEY (Codigo_Asignacion),
    CONSTRAINT FK_Asignaciones_Choferes FOREIGN KEY (Codigo_Chofer)
        REFERENCES Choferes (Codigo_Chofer),
    CONSTRAINT FK_Asignaciones_Colectivos FOREIGN KEY (Nro_Coche)
        REFERENCES Colectivos (Nro_Coche)
);
GO

/* 5. CALLES, PARADAS Y RECORRIDOS */

-- Paradas_Calles se elimina. Cada parada referencia una calle.
CREATE TABLE Paradas (
    Codigo_Parada INT IDENTITY(1,1) NOT NULL,
    Descripcion VARCHAR(150) NULL,
    Codigo_Calle INT NOT NULL,
    Numero_Calle VARCHAR(10) NULL,
    CONSTRAINT PK_Paradas PRIMARY KEY (Codigo_Parada),
    CONSTRAINT FK_Paradas_Calles FOREIGN KEY (Codigo_Calle)
        REFERENCES Calles (Codigo_Calle)
);
GO

CREATE TABLE Recorridos (
    Codigo_Recorrido INT IDENTITY(1,1) NOT NULL,
    Descripcion VARCHAR(150) NULL,
    Codigo_Linea INT NOT NULL,
    Codigo_Sentido INT NOT NULL,
    CONSTRAINT PK_Recorridos PRIMARY KEY (Codigo_Recorrido),
    CONSTRAINT FK_Recorridos_Lineas FOREIGN KEY (Codigo_Linea)
        REFERENCES Lineas (Codigo_Linea),
    CONSTRAINT FK_Recorridos_Sentidos FOREIGN KEY (Codigo_Sentido)
        REFERENCES Sentidos_Viaje (Codigo_Sentido)
);
GO

CREATE TABLE Recorridos_Paradas (
    Codigo_Recorrido INT NOT NULL,
    Codigo_Parada INT NOT NULL,
    Orden INT NOT NULL,
    CONSTRAINT PK_Recorridos_Paradas PRIMARY KEY (Codigo_Recorrido, Codigo_Parada),
    CONSTRAINT FK_Recorridos_Paradas_Recorridos FOREIGN KEY (Codigo_Recorrido)
        REFERENCES Recorridos (Codigo_Recorrido),
    CONSTRAINT FK_Recorridos_Paradas_Paradas FOREIGN KEY (Codigo_Parada)
        REFERENCES Paradas (Codigo_Parada),
    CONSTRAINT UQ_Recorridos_Paradas_Orden UNIQUE (Codigo_Recorrido, Orden)
);
GO

/* 6. MANTENIMIENTOS
   Correccion: Nro_Coche es FK hacia Colectivos.Nro_Coche. */

CREATE TABLE Mantenimientos (
    Codigo_Mantenimiento INT IDENTITY(1,1) NOT NULL,
    Nro_Coche INT NOT NULL,
    Codigo_Tipo_Mantenimiento INT NOT NULL,
    Fecha DATE NOT NULL,
    Descripcion VARCHAR(250) NULL,
    Costo DECIMAL(12,2) NULL,
    CONSTRAINT PK_Mantenimientos PRIMARY KEY (Codigo_Mantenimiento),
    CONSTRAINT FK_Mantenimientos_Colectivos FOREIGN KEY (Nro_Coche)
        REFERENCES Colectivos (Nro_Coche),
    CONSTRAINT FK_Mantenimientos_Tipos FOREIGN KEY (Codigo_Tipo_Mantenimiento)
        REFERENCES Tipos_Mantenimiento (Codigo_Tipo_Mantenimiento)
);
GO

/* 7. SERVICIOS Y UBICACION DEL COCHE */

CREATE TABLE Servicios (
    Codigo_Servicio INT IDENTITY(1,1) NOT NULL,
    Codigo_Recorrido INT NOT NULL,
    Nro_Coche INT NOT NULL,
    Codigo_Chofer INT NOT NULL,
    Fecha DATE NOT NULL,
    Hora_inicio TIME(0) NOT NULL,
    Hora_Fin TIME(0) NOT NULL,
    CONSTRAINT PK_Servicios PRIMARY KEY (Codigo_Servicio),
    CONSTRAINT FK_Servicios_Recorridos FOREIGN KEY (Codigo_Recorrido)
        REFERENCES Recorridos (Codigo_Recorrido),
    CONSTRAINT FK_Servicios_Colectivos FOREIGN KEY (Nro_Coche)
        REFERENCES Colectivos (Nro_Coche),
    CONSTRAINT FK_Servicios_Choferes FOREIGN KEY (Codigo_Chofer)
        REFERENCES Choferes (Codigo_Chofer)
);
GO

CREATE TABLE Ubicacion_Coche (
    Codigo_Ubicacion_Coche INT IDENTITY(1,1) NOT NULL,
    Codigo_Servicio INT NOT NULL,
    Latitud DECIMAL(9,6) NULL,
    Longitud DECIMAL(9,6) NULL,
    Fecha_Hora DATETIME2(0) NOT NULL,
    Velocidad_KmxH DECIMAL(7,2) NULL,
    CONSTRAINT PK_Ubicacion_Coche PRIMARY KEY (Codigo_Ubicacion_Coche),
    CONSTRAINT FK_Ubicacion_Coche_Servicios FOREIGN KEY (Codigo_Servicio)
        REFERENCES Servicios (Codigo_Servicio)
);
GO

/* 8. BOLETOS Y FORMA DE PAGO
   Se mantiene Importe como valor cobrado al vender el boleto. */

CREATE TABLE Boletos (
    Codigo_Boleto INT IDENTITY(1,1) NOT NULL,
    Codigo_Servicio INT NOT NULL,
    Codigo_Tipo_Boleto INT NOT NULL,
    Fecha_Hora DATETIME2(0) NOT NULL,
    Importe DECIMAL(10,2) NOT NULL,
    Codigo_Forma_Pago INT NOT NULL,
    CONSTRAINT PK_Boletos PRIMARY KEY (Codigo_Boleto),
    CONSTRAINT FK_Boletos_Servicios FOREIGN KEY (Codigo_Servicio)
        REFERENCES Servicios (Codigo_Servicio),
    CONSTRAINT FK_Boletos_Tipos_Boleto FOREIGN KEY (Codigo_Tipo_Boleto)
        REFERENCES Tipos_Boleto (Codigo_Tipo_Boleto),
    CONSTRAINT FK_Boletos_Formas_Pago FOREIGN KEY (Codigo_Forma_Pago)
        REFERENCES Formas_Pago (Codigo_Forma_Pago)
);
GO

/* 9. REPUESTOS Y CONTROL DE STOCK */

CREATE TABLE Repuestos (
    Codigo_Repuesto INT IDENTITY(1,1) NOT NULL,
    Descripcion VARCHAR(120) NOT NULL,
    Codigo_Tipo_Repuesto INT NOT NULL,
    Codigo_Proveedor INT NOT NULL,
    Stock_Minimo INT NOT NULL,
    Precio_Unitario DECIMAL(12,2) NULL,
    CONSTRAINT PK_Repuestos PRIMARY KEY (Codigo_Repuesto),
    CONSTRAINT FK_Repuestos_Tipos FOREIGN KEY (Codigo_Tipo_Repuesto)
        REFERENCES Tipos_Repuesto (Codigo_Tipo_Repuesto),
    CONSTRAINT FK_Repuestos_Proveedores FOREIGN KEY (Codigo_Proveedor)
        REFERENCES Proveedores (Codigo_Proveedor)
);
GO

CREATE TABLE Movimientos_Stock (
    Codigo_Movimiento INT IDENTITY(1,1) NOT NULL,
    Codigo_Repuesto INT NOT NULL,
    Fecha DATETIME2(0) NOT NULL,
    Tipo_Movimiento VARCHAR(10) NOT NULL,
    Cantidad INT NOT NULL,
    Observacion VARCHAR(250) NULL,
    CONSTRAINT PK_Movimientos_Stock PRIMARY KEY (Codigo_Movimiento),
    CONSTRAINT FK_Movimientos_Stock_Repuestos FOREIGN KEY (Codigo_Repuesto)
        REFERENCES Repuestos (Codigo_Repuesto)
);
GO

/* 10. DETALLE DE REPUESTOS UTILIZADOS EN MANTENIMIENTOS */

CREATE TABLE Detalle_Mantenimiento (
    Codigo_Detalle_Mantenimiento INT IDENTITY(1,1) NOT NULL,
    Codigo_Mantenimiento INT NOT NULL,
    Codigo_Repuesto INT NOT NULL,
    Cantidad INT NOT NULL,
    Costo_Unitario DECIMAL(12,2) NOT NULL,
    CONSTRAINT PK_Detalle_Mantenimiento PRIMARY KEY (Codigo_Detalle_Mantenimiento),
    CONSTRAINT FK_Detalle_Mantenimiento_Mantenimientos FOREIGN KEY (Codigo_Mantenimiento)
        REFERENCES Mantenimientos (Codigo_Mantenimiento),
    CONSTRAINT FK_Detalle_Mantenimiento_Repuestos FOREIGN KEY (Codigo_Repuesto)
        REFERENCES Repuestos (Codigo_Repuesto)
);
GO

/* ------------------------------------------------------------
   CONSULTA 1 (subconsulta con NOT EXISTS)
   Enunciado: Listar los colectivos que no tuvieron ningun mantenimiento
   en los ultimos 6 meses, indicando la linea a la que pertenecen.

   Utilidad: le permite al jefe de mantenimiento (encargado del taller)
   detectar que unidades de la flota estan atrasadas con sus controles
   y programar primero los services de esos coches, sabiendo ademas a que
   linea pertenece cada uno para coordinar su salida de servicio.
------------------------------------------------------------ */
SELECT  co.Nro_Coche, co.Patente, co.Marca, co.Modelo, l.Numero_Linea
FROM Colectivos co
JOIN Lineas l ON l.Codigo_Linea = co.Codigo_Linea
WHERE NOT EXISTS (
        SELECT 1
        FROM Mantenimientos m
        WHERE m.Nro_Coche = co.Nro_Coche
          AND m.Fecha >= DATEADD(MONTH, -6, GETDATE())
)
ORDER BY l.Numero_Linea, co.Nro_Coche;
GO


/* ------------------------------------------------------------
Consulta 2: Procedimiento almacenado para consultar servicios por periodo

Crear un procedimiento almacenado que reciba una fecha de inicio y una fecha de fin como parametros, y muestre todos los servicios realizados dentro de ese periodo. El resultado debe incluir la fecha del servicio, el horario de inicio y finalizacion, el nombre y apellido del chofer, el numero de coche, la patente, el numero de linea y la descripcion del recorrido. Los resultados deben mostrarse ordenados por fecha y horario de inicio.

Utilidad: le permite al encargado de operaciones (despachante) revisar rapidamente la actividad de cualquier periodo, por ejemplo para controlar que chofer condujo cada coche, verificar el cumplimiento de los horarios o responder un reclamo sobre un servicio puntual, sin tener que armar la consulta cada vez.
   ------------------------------------------------------------ */

CREATE OR ALTER PROCEDURE ConsultarServiciosPorPeriodo
    @FechaInicio DATE,
    @FechaFin DATE
AS
BEGIN
    SELECT
        S.Fecha,
        S.Hora_inicio,
        S.Hora_Fin,
        CH.Nombre,
        CH.Apellido,
        C.Nro_Coche,
        C.Patente,
        L.Numero_Linea,
        R.Descripcion AS Recorrido
    FROM Servicios S
    INNER JOIN Choferes CH
        ON S.Codigo_Chofer = CH.Codigo_Chofer
    INNER JOIN Colectivos C
        ON S.Nro_Coche = C.Nro_Coche
    INNER JOIN Recorridos R
        ON S.Codigo_Recorrido = R.Codigo_Recorrido
    INNER JOIN Lineas L
        ON R.Codigo_Linea = L.Codigo_Linea
    WHERE S.Fecha BETWEEN @FechaInicio AND @FechaFin
    ORDER BY S.Fecha, S.Hora_inicio;
END;
GO

EXEC ConsultarServiciosPorPeriodo
    @FechaInicio = '2026-10-01',
    @FechaFin = '2026-10-10';
GO

/* ------------------------------------------------------------
Consulta 3: Vista de recaudacion por linea y tipo de boleto en los ultimos 3 meses.

La vista permite analizar la actividad economica de cada linea de transporte durante los ultimos tres meses. Para ello, muestra el numero de linea, el tipo de boleto, la cantidad de boletos registrados y la recaudacion total. Para obtener estos resultados, se relacionan las tablas Boletos, Servicios, Recorridos, Lineas y Tipos_Boleto. Se utilizan funciones de agregacion y GROUP BY para agrupar los datos por linea y tipo de boleto, mientras que WHERE y DATEADD permiten limitar el analisis al periodo establecido.

Utilidad: le permite al gerente comercial (o al responsable de finanzas) comparar cuanto recauda cada linea y que peso tiene cada tipo de boleto, incluyendo los boletos sin costo (BEC y BAM). Con eso puede evaluar la rentabilidad de cada linea y el impacto de los beneficios gratuitos en los ingresos.
   ------------------------------------------------------------ */

CREATE OR ALTER VIEW Vista_Recaudacion_Linea_Tipo_Boleto
AS
SELECT
    L.Numero_Linea,
    TB.Descripcion AS Tipo_Boleto,
    COUNT(B.Codigo_Boleto) AS Cantidad_Boletos,
    SUM(B.Importe) AS Recaudacion_Total
FROM Boletos B
INNER JOIN Servicios S
    ON B.Codigo_Servicio = S.Codigo_Servicio
INNER JOIN Recorridos R
    ON S.Codigo_Recorrido = R.Codigo_Recorrido
INNER JOIN Lineas L
    ON R.Codigo_Linea = L.Codigo_Linea
INNER JOIN Tipos_Boleto TB
    ON B.Codigo_Tipo_Boleto = TB.Codigo_Tipo_Boleto
WHERE S.Fecha >= DATEADD(MONTH, -3, GETDATE())
GROUP BY
    L.Numero_Linea,
    TB.Descripcion;
GO

SELECT * FROM Vista_Recaudacion_Linea_Tipo_Boleto ORDER BY Numero_Linea, Tipo_Boleto;
GO

/* ------------------------------------------------------------
Consulta 4: Lineas con una cantidad de boletos superior al promedio.

La consulta permite identificar las lineas que registraron una mayor cantidad de boletos en comparacion con el promedio general. Para ello, se relacionan las tablas correspondientes mediante JOIN, se utiliza COUNT para contar los boletos de cada linea y una subconsulta para calcular el promedio de esas cantidades. Finalmente, se utiliza HAVING para mostrar unicamente las lineas que superan dicho promedio.

Utilidad: le permite al gerente general o responsable de planificacion de la red identificar las lineas con mayor demanda de pasajeros, para decidir donde conviene reforzar frecuencias, asignar mas coches o priorizar inversiones.
------------------------------------------------------------ */
SELECT
    L.Numero_Linea,
    COUNT(B.Codigo_Boleto) AS Cantidad_Boletos,
    SUM(B.Importe) AS Recaudacion_Total,
    AVG(B.Importe) AS Importe_Promedio
FROM Lineas L
INNER JOIN Recorridos R
    ON L.Codigo_Linea = R.Codigo_Linea
INNER JOIN Servicios S
    ON R.Codigo_Recorrido = S.Codigo_Recorrido
INNER JOIN Boletos B
    ON S.Codigo_Servicio = B.Codigo_Servicio
GROUP BY
    L.Numero_Linea
HAVING COUNT(B.Codigo_Boleto) >
(
    SELECT AVG(Cantidad_Boletos * 1.0)
    FROM
    (
        SELECT
            L2.Codigo_Linea,
            COUNT(B2.Codigo_Boleto) AS Cantidad_Boletos
        FROM Lineas L2
        INNER JOIN Recorridos R2
            ON L2.Codigo_Linea = R2.Codigo_Linea
        INNER JOIN Servicios S2
            ON R2.Codigo_Recorrido = S2.Codigo_Recorrido
        INNER JOIN Boletos B2
            ON S2.Codigo_Servicio = B2.Codigo_Servicio
        GROUP BY L2.Codigo_Linea
    ) AS PromedioLineas
)
ORDER BY Cantidad_Boletos DESC;
GO


/* ------------------------------------------------------------
Consulta 5: Estado de las asignaciones de los choferes

Esta vista permite consultar la cantidad de asignaciones registradas para cada chofer, identificando cuales tienen asignaciones y cuales no. Utiliza las tablas Choferes y Asignaciones, combinando INNER JOIN, GROUP BY, COUNT, NOT EXISTS y UNION ALL para clasificar a los choferes segun su situacion actual. De esta manera, facilita el seguimiento de las asignaciones y permite identificar a los choferes que no tienen ninguna asignacion registrada.

Utilidad: le permite al jefe de personal (encargado de la programacion de choferes) ver cuantas asignaciones y cuantas horas lleva cada chofer, para repartir la carga de trabajo de forma pareja y detectar a los choferes que estan sin asignar y pueden cubrir ausencias o nuevos servicios.
------------------------------------------------------------ */

CREATE OR ALTER VIEW Vista_Estado_Asignaciones_Choferes AS
    SELECT C.Codigo_Chofer,
        C.Nombre,
        C.Apellido,
        'Con asignaciones' AS Estado,
        COUNT(A.Codigo_Asignacion) AS Cantidad_Asignaciones,
        SUM( DATEDIFF(MINUTE, A.Hora_inicio, A.Hora_Fin) ) AS Minutos_Asignados
    FROM Choferes C
        INNER JOIN Asignaciones A ON C.Codigo_Chofer = A.Codigo_Chofer
    GROUP BY C.Codigo_Chofer, C.Nombre, C.Apellido

    UNION ALL

    SELECT C.Codigo_Chofer,
        C.Nombre,
        C.Apellido,
        'Sin asignaciones' AS Estado,
        0 AS Cantidad_Asignaciones,
        0 AS Minutos_Asignados
    FROM Choferes C
    WHERE NOT EXISTS
        ( SELECT 1
        FROM Asignaciones A
        WHERE A.Codigo_Chofer = C.Codigo_Chofer );
GO

SELECT *
FROM Vista_Estado_Asignaciones_Choferes
ORDER BY Estado, Codigo_Chofer;
GO
